% C++ corners (tools/bookfilter.py): a light box with a rule on its
% left, so that they don't look like quotations. Like the shading of code
% blocks, the box extends past the text on both sides, so the corner's text
% keeps the full width. Code blocks in a corner have no shading of their
% own (it takes the box's color), since the box already sets them apart.
% The box is a pale tint of the links' blue, and the rule a stronger one,
% chosen so that in gray (0.3 R + 0.59 G + 0.11 B, which xcolor uses, and
% the gray edition selects) they become exactly the grays 0.96 and 0.55.
\definecolor{cornerback}{rgb}{0.948,0.963,0.977}
\definecolor{cornerrule}{rgb}{0.42,0.581,0.742}
\newtcolorbox{cppcorner}{breakable, colframe=cornerrule, colback=cornerback, sharp corners,
  boxrule=0pt, leftrule=1.5pt,
  left=3.5mm, right=3mm, top=2mm, bottom=2mm, grow to left by=5mm, grow to right by=3mm,
  before skip=\medskipamount, after skip=\medskipamount,
  before upper={\colorlet{shadecolor}{cornerback}\parskip=\medskipamount}}
% Other sidebars, such as "About the name": the same box in gray, a shade

\newtcolorbox{sidebar}{breakable, colframe=black!45, colback=black!4, sharp corners,

  left=3.5mm, right=3mm, top=2mm, bottom=2mm, grow to left by=5mm, grow to right by=3mm,

  before upper={\colorlet{shadecolor}{cornerback}\parskip=\medskipamount}}
% A code block taken from a file (tools/bookfilter.py) is labeled with the
% file's path, small and gray, just above the block, as a caption, ending at
% the right edge of the shading (which extends \fboxsep past the text). If
% fewer than six lines are left on the page (the label, and a few lines of
% code), the label goes to the next
% page with the code: the test of LaTeX's needspace package, which compares
% the space left on the page with what's needed, and breaks the page if it
% isn't there. (A stretchable break before the label, which we used first,
% isn't enough: when the code didn't fit at all, framed.sty ended the page
% after the label, alone.) The shaded box below adds \topsep above itself
% (framed.sty's \OuterFrameSep), which can stretch and shrink; on a page
% that LaTeX squeezes to fit, it would bring the label down onto the box.
% So the space here cancels \topsep's shrink, and the gap
% between the label and the box is never smaller than it should be.
% Finally, \@nobreaktrue, as after a section heading: framed.sty's
% \MakeFramed adds breakpoints of its own above the box (\penalty-30 and
% \penalty0) unless it's set.
\makeatletter
\newcommand{\codefile}[1]{%
  \par\penalty-100
  \begingroup
    \dimen@=6\baselineskip
    \dimen@ii=\pagegoal \advance\dimen@ii by -\pagetotal
    \ifdim\dimen@>\dimen@ii \ifdim\dimen@ii>\z@ \vfil\fi \break\fi
  \endgroup
  \addvspace{\smallskipamount}%
  \noindent\makebox[\linewidth][r]{\footnotesize\color{black!55}\texttt{#1}\hspace{-\fboxsep}}\par
  \nopagebreak
  \vskip -0.9\baselineskip minus -\glueshrink\topsep
  \nopagebreak\@nobreaktrue}
\makeatother

% The font of the code names in the index (tools/bookfilter.py): the

  FakeStretch=0.85]
