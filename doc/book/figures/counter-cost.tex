% Figure for chapter 13, "Sharing anyway": counter.cc's time per increment
% on each shard, with one counter for each shard and with one shared
% counter, on 1, 2 and 4 of the laptop's fast cores (the middle of three
% runs' range: own 6.1, 7.25, 7.85 ns; shared 6.1, 27.75, 65.05 ns).
\begin{tikzpicture}[x=1mm, y=1mm]
  % 1 ns is 0.8 mm; the shards are 1, 2 and 4, evenly spaced.
  \def\xa{12} \def\xb{42} \def\xc{72}
  % Axes and grid.
  \foreach \v in {0, 20, 40, 60} {
    \draw[draw=fignote!40, line width=0.3pt] (0, \v*0.8) -- (84, \v*0.8);
    \node[note, anchor=east] at (-1, \v*0.8) {\v};
  }
  \draw[flow] (0, 0) -- (0, 57.60);
  \draw[draw=figink] (0, 0) -- (84, 0);
  \node[note, rotate=90, anchor=south] at (-9, 27.20) {ns per increment};
  \foreach \x/\n in {\xa/1, \xb/2, \xc/4}
    \node[note, anchor=north] at (\x, -1) {\n};
  \node[note, anchor=north] at (42, -6) {shards};

  % One shared counter: squares, on a dashed line.
  \draw[draw=fighot, line width=0.9pt, dash pattern=on 3pt off 2pt]
    (\xa, 4.88) -- (\xb, 22.20) -- (\xc, 52.04);
  \foreach \x/\v in {\xa/6.1, \xb/27.75, \xc/65.05}
    \fill[fighot] (\x-1.3, \v*0.8-1.3) rectangle (\x+1.3, \v*0.8+1.3);
  \node[font=\sffamily\footnotesize, text=fighot, anchor=west, align=left] at (\xc+2.5, 52.04)
    {one shared \\ counter};

  % Its own counter on each shard: circles, on a solid line.
  \draw[draw=figcool, line width=0.9pt]
    (\xa, 4.88) -- (\xb, 5.80) -- (\xc, 6.28);
  \foreach \x/\v in {\xa/6.1, \xb/7.25, \xc/7.85}
    \draw[draw=figcool, fill=white, line width=0.9pt] (\x, \v*0.8) circle (1.4);
  \node[font=\sffamily\footnotesize, text=figcool, anchor=west, align=left] at (\xc+2.5, 7.28)
    {its own counter \\ on each shard};
\end{tikzpicture}
