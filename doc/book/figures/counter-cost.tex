% Figure for chapter 13, "Sharing anyway": counter.cc's time per increment
% on each shard, with one plain counter for each shard and with one shared
% atomic counter, on 1 to 4 of the laptop's fast cores (the best of five
% runs: per-thread 0.9, 0.9, 1.0, 1.0 ns; shared 7.4, 26.2, 45.0, 62.9 ns).
\begin{tikzpicture}[x=1mm, y=1mm]
  % 1 ns is 0.8 mm; the shards are 1 to 4, evenly spaced.
  \def\xa{12} \def\xb{32} \def\xc{52} \def\xd{72}
  % Axes and grid.
  \foreach \v in {0, 20, 40, 60} {
    \draw[draw=fignote!40, line width=0.3pt] (0, \v*0.8) -- (84, \v*0.8);
    \node[axis, anchor=east] at (-1, \v*0.8) {\v};
  }
  \draw[flow] (0, 0) -- (0, 57.60);
  \draw[draw=figink] (0, 0) -- (84, 0);
  \node[axis, rotate=90, anchor=south] at (-9, 27.20) {ns per increment};
  \foreach \x/\n in {\xa/1, \xb/2, \xc/3, \xd/4}
    \node[axis, anchor=north] at (\x, -1) {\n};
  \node[axis, anchor=north] at (42, -7) {shards};

  % One shared counter: squares, on a dashed line.
  \draw[draw=fighot, line width=0.9pt, dash pattern=on 3pt off 2pt]
    (\xa, 5.92) -- (\xb, 20.96) -- (\xc, 36.00) -- (\xd, 50.32);
  \foreach \x/\v in {\xa/7.4, \xb/26.2, \xc/45.0, \xd/62.9}
    \fill[fighot] (\x-1.3, \v*0.8-1.3) rectangle (\x+1.3, \v*0.8+1.3);
  % Above the grid line at 60 ns, which is at 48 mm.
  \node[font=\sffamily\footnotesize, text=fighot, anchor=south west, align=left] at (\xd+2.5, 50)
    {one shared counter};

  % A per-thread counter: circles, on a solid line.
  \draw[draw=figcool, line width=0.9pt]
    (\xa, 0.72) -- (\xb, 0.72) -- (\xc, 0.80) -- (\xd, 0.80);
  \foreach \x/\v in {\xa/0.9, \xb/0.9, \xc/1.0, \xd/1.0}
    \draw[draw=figcool, fill=white, line width=0.9pt] (\x, \v*0.8) circle (1.4);
  \node[font=\sffamily\footnotesize, text=figcool, anchor=west, align=left] at (\xd+2.5, 3.5)
    {per-thread \\ counter};
\end{tikzpicture}
